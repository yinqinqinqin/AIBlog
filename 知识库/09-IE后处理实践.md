#### shader实现

```glsl
vec3 splitRotateCol(float start, float end, float scale,vec2 uv){
    //start:起始uv end：结束uv scale：时间因子 uv：纹理坐标
    vec3 col = vec3(0.0);
    float factor1 = start + (1.0 - scale)*(end - start)/2.0;//求edge1
    float factor2 = end -(1.0-scale)*(end - start)/2.0;//求edge2

    uv.x = uv.x * (step(factor1,uv.x) - step(factor2 ,uv.x));//通过2个edge去构造门函数在X方向remap
    float scaleY = mix(1.0,1.2,(uv.x - factor1)/(factor2 - factor1));//随X坐标变化的y方向缩放，制造伪3d
    scaleY = mix(scaleY,1.0,scale);//根据时间插值，0开始，1结束到达原图
    uv.y = (uv.y - 0.5) *scaleY + 0.5;
    col = texture(iChannel0,uv).rgb ;
    col *= (step(0.0,uv.y) - step(1.0,uv.y)) * step(0.0001,uv.x);

    return col;
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    vec2 uv = fragCoord/iResolution.xy;
    float speed = mix(1.3,3.5,1.0 -  abs(2.0 * (uv.x - 0.5)));//从中间到两侧，控制转场的速度
    float scale = pow(clamp(0.0,1.0,iTime/1.3),2.0) * speed;//二次曲线，系数越大越快达到1

    scale = clamp(0.0,1.0,scale);
    vec3 sum = vec3(0.0);
    int n = 30;
    for(int i = 0;i < n;i++){
        sum += splitRotateCol(float(i)/float(n),float(i+1)/float(n),scale,uv);
    }
    fragColor = vec4(sum,1.0);
}
```
